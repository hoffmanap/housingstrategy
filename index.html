function runSimulation() { // 1. Gather UI Inputs const scenario = document.getElementById('scenario_select').value; const distFilter = document.getElementById('filter_district').value; const transitOnly = document.getElementById('filter_transit').checked; const protectHistoric = document.getElementById('filter_historic').checked; const absorption = parseFloat(document.getElementById('absorption_rate').value) / 100; const adus = document.getElementById('allow_adus').checked; const lotSplits = document.getElementById('allow_lot_splits').checked; const mm48 = document.getElementById('mm_4_8').checked; const mm816 = document.getElementById('mm_8_16').checked; const midrise = document.getElementById('allow_midrise').checked; const midriseFloors = parseInt(document.getElementById('midrise_floors').value); // Accumulate floats in these objects to prevent rounding errors let districtTotals = { 1:{u:0,j:0}, 2:{u:0,j:0}, 3:{u:0,j:0}, 4:{u:0,j:0}, 5:{u:0,j:0}, 6:{u:0,j:0}, 7:{u:0,j:0}, 8:{u:0,j:0} }; allParcels.forEach(parcel => { const props = parcel.properties; const d = props.district; props._isVisible = true; // FILTERS if (distFilter !== 'all' && d.toString() !== distFilter) props._isVisible = false; if (transitOnly && props.in_corridor !== true) props._isVisible = false; if (protectHistoric && props.in_historic === true) props._isVisible = false; let parcelUnits = 0; let parcelJobs = 0; if (props._isVisible) { const zoning = (props.zonelabel || '').toUpperCase(); let isRes = zoning.includes('R') || zoning.includes('A') || zoning.includes('M') || zoning.includes('P'); let isCom = zoning.includes('C'); // Scenario Overrides if (scenario === 'residential') { isRes = true; isCom = false; } else if (scenario === 'commercial') { isRes = false; isCom = true; } else if (scenario === 'mixed') { isRes = true; isCom = true; } // --- HOUSING CALCULATION --- if (isRes) { if (adus) parcelUnits += 1; if (lotSplits && props.targetlotarea > 8000) parcelUnits += 2; if (mm48) parcelUnits += 6; if (mm816) parcelUnits += 12; } // --- JOBS CALCULATION --- if (isCom) { // BASELINE: All commercial parcels have a baseline job capacity // Even if Midrise is unchecked, we assume some baseline job density parcelJobs += 5; // POLICY: Midrise adds additional capacity if (midrise) { parcelJobs += 15; parcelUnits += (midriseFloors - 1) * 5; } } // Accumulate RAW floats (Do not round here) districtTotals[d].u += (parcelUnits * absorption); districtTotals[d].j += (parcelJobs * absorption); // Store for map styling props._calcUnits = (parcelUnits * absorption); props._calcJobs = (parcelJobs * absorption); } }); // 3. Update Results Table (Rounding only happens here) const tbody = document.getElementById('results-body'); tbody.innerHTML = ''; let grandUnits = 0; let grandJobs = 0; for (let i = 1; i <= 8; i++) { if (distFilter !== 'all' && i.toString() !== distFilter) continue; const u = Math.round(districtTotals[i].u); const j = Math.round(districtTotals[i].j); grandUnits += u; grandJobs += j; tbody.innerHTML += `D${i}${u.toLocaleString()}${j.toLocaleString()}`; } document.getElementById('total-units').innerText = grandUnits.toLocaleString(); document.getElementById('total-jobs').innerText = grandJobs.toLocaleString(); // 4. Update Map geojsonLayer.eachLayer(layer => { const props = layer.feature.properties; if (!props._isVisible) { layer.setStyle({ opacity: 0, fillOpacity: 0 }); return; } const u = props._calcUnits || 0; const j = props._calcJobs || 0; // Style based on dominant use layer.setStyle({ fillColor: (j > u) ? "#d35400" : (u > j ? "#2980b9" : "#bdc3c7"), fillOpacity: (u+j > 0) ? 0.6 : 0.1, stroke: true, opacity: 1, weight: 0.5 }); }); }
